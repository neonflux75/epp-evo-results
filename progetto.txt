def http_get_json(url, data=None, timeout=35):
    headers = {"User-Agent": "EvoResultsTrackGenerator/2.0"}
    req = urllib.request.Request(url, data=data, headers=headers)
    with urllib.request.urlopen(req, timeout=timeout) as resp:
        return json.loads(resp.read().decode("utf-8"))


def fetch_from_f1_github(f1_id):
    if not f1_id:
        return None
    url = f"{F1_CIRCUITS_BASE}{f1_id}.geojson"
    try:
        geo = http_get_json(url, timeout=15)
        features = geo.get("features", [])
        if not features:
            return None
        geom = features[0].get("geometry", {})
        coords = geom.get("coordinates", [])
        if geom.get("type") == "MultiLineString":
            coords = max(coords, key=len)
        return [(pt[0], pt[1]) for pt in coords if len(pt) >= 2]
    except Exception:
        return None


def stitch_segments(segments):
    if not segments:
        return []
    if len(segments) == 1:
        return segments[0]

    # Filtra segmenti troppo corti (pitlane/raccordi secondari) se ci sono segmenti principali
    segments = sorted(segments, key=len, reverse=True)
    chain = list(segments[0])
    remaining = [list(s) for s in segments[1:]]

    def dist(p1, p2):
        return math.hypot(p1[0] - p2[0], p1[1] - p2[1])

    # Collega in catena i segmenti contigui entro una soglia ragionevole (~350m in gradi ~0.0035)
    MAX_GAP = 0.0045
    changed = True
    while changed and remaining:
        changed = False
        best_idx = -1
        best_mode = None
        best_d = MAX_GAP

        for i, seg in enumerate(remaining):
            d_end_start = dist(chain[-1], seg[0])
            d_end_end = dist(chain[-1], seg[-1])
            d_start_end = dist(chain[0], seg[-1])
            d_start_start = dist(chain[0], seg[0])

            candidates = [
                (d_end_start, "append"),
                (d_end_end, "append_rev"),
                (d_start_end, "prepend"),
                (d_start_start, "prepend_rev"),
            ]
            for d, mode in candidates:
                if d < best_d:
                    best_d = d
                    best_idx = i
                    best_mode = mode

        if best_idx != -1:
            seg = remaining.pop(best_idx)
            if best_mode == "append":
                chain.extend(seg[1:])
            elif best_mode == "append_rev":
                chain.extend(list(reversed(seg))[1:])
            elif best_mode == "prepend":
                chain = seg[:-1] + chain
            elif best_mode == "prepend_rev":
                chain = list(reversed(seg))[:-1] + chain
            changed = True

    return chain


def fetch_from_overpass(coords_hint, query_name):
    if coords_hint:
        lat, lon, radius = coords_hint
        ql = f"""
        [out:json][timeout:30];
        (
          relation["route"="raceway"](around:{radius},{lat},{lon});
          way["highway"="raceway"](around:{radius},{lat},{lon});
        );
        out body;
        >;
        out skel qt;
        """
    else:
        safe_name = query_name.replace('"', '\"')
        ql = f"""
        [out:json][timeout:30];
        (
          relation["route"="raceway"]["name"~"{safe_name}",i];
          way["highway"="raceway"]["name"~"{safe_name}",i];
        );
        out body;
        >;
        out skel qt;
        """

    payload = urllib.parse.urlencode({"data": ql}).encode("utf-8")

    for mirror in OVERPASS_MIRRORS:
        for attempt in range(2):
            try:
                res = http_get_json(mirror, data=payload, timeout=35)
                elements = res.get("elements", [])
                if not elements:
                    break

                nodes = {}
                ways = {}
                relations = []

                for el in elements:
                    t = el.get("type")
                    if t == "node":
                        nodes[el["id"]] = (el["lon"], el["lat"])
                    elif t == "way":
                        ways[el["id"]] = el
                    elif t == "relation":
                        relations.append(el)

                # Preferisci una relazione chiusa (es. circuito principale) se disponibile
                target_way_ids = []
                if relations:
                    best_rel = max(relations, key=lambda r: len(r.get("members", [])))
                    for m in best_rel.get("members", []):
                        if m.get("type") == "way" and m.get("role", "") in ("", "forward", "backward"):
                            if m["ref"] in ways:
                                target_way_ids.append(m["ref"])

                if not target_way_ids:
                    for wid, w in ways.items():
                        tags = w.get("tags", {})
                        # Escludi pitlane o kartodromi se c'è la pista principale
                        if tags.get("service") in ("pit_lane", "parking_aisle"):
                            continue
                        target_way_ids.append(wid)

                segments = []
                for wid in target_way_ids:
                    w = ways.get(wid)
                    if not w:
                        continue
                    pts = [nodes[nid] for nid in w.get("nodes", []) if nid in nodes]
                    if len(pts) >= 2:
                        segments.append(pts)

                stitched = stitch_segments(segments)
                if len(stitched) >= 15:
                    return stitched
            except Exception as e:
                print(f"   [Mirror {mirror.split('/')[2]} tentativo {attempt+1}] {e}")
                time.sleep(2.5)
    return None


def render_track_png(points, out_paths):
    W, H = 640, 360
    PADDING = 44

    # Proiezione equirettangolare corretta per latitudine media
    avg_lat = sum(p[1] for p in points) / len(points)
    cos_lat = math.cos(math.radians(avg_lat))
    proj = [(p[0] * cos_lat, -p[1]) for p in points]

    min_x = min(p[0] for p in proj)
    max_x = max(p[0] for p in proj)
    min_y = min(p[1] for p in proj)
    max_y = max(p[1] for p in proj)

    span_x = max(max_x - min_x, 1e-6)
    span_y = max(max_y - min_y, 1e-6)

    scale = min((W - 2 * PADDING) / span_x, (H - 2 * PADDING) / span_y)
    offset_x = (W - span_x * scale) / 2.0
    offset_y = (H - span_y * scale) / 2.0

    canvas_pts = [
        ((p[0] - min_x) * scale + offset_x, (p[1] - min_y) * scale + offset_y)
        for p in proj
    ]

    # Sfondo scuro telemetrico trasparente/scuro
    img = Image.new("RGBA", (W, H), (14, 17, 24, 255))

    # Griglia telemetrica sottile
    grid = Image.new("RGBA", (W, H), (0, 0, 0, 0))
    gdraw = ImageDraw.Draw(grid)
    for gx in range(0, W, 32):
        gdraw.line([(gx, 0), (gx, H)], fill=(255, 255, 255, 8), width=1)
    for gy in range(0, H, 32):
        gdraw.line([(0, gy), (W, gy)], fill=(255, 255, 255, 8), width=1)
    img = Image.alpha_composite(img, grid)

    # Alone rosso esterno
    glow = Image.new("RGBA", (W, H), (0, 0, 0, 0))
    glow_draw = ImageDraw.Draw(glow)
    glow_draw.line(canvas_pts, fill=(239, 68, 68, 120), width=14, joint="curve")
    glow = glow.filter(ImageFilter.GaussianBlur(radius=6))
    img = Image.alpha_composite(img, glow)

    # Tracciato principale bianco/rosso ad alto contrasto
    track_layer = Image.new("RGBA", (W, H), (0, 0, 0, 0))
    tdraw = ImageDraw.Draw(track_layer)
    tdraw.line(canvas_pts, fill=(30, 35, 48, 255), width=11, joint="curve")
    tdraw.line(canvas_pts, fill=(241, 245, 249, 255), width=5, joint="curve")

    # Linea del traguardo (Start/Finish) perpendicolare al primo segmento
    if len(canvas_pts) >= 2:
        p0, p1 = canvas_pts[0], canvas_pts[1]
        dx, dy = p1[0] - p0[0], p1[1] - p0[1]
        length = math.hypot(dx, dy)
        if length > 0:
            nx, ny = -dy / length, dx / length
            tdraw.line(
                [(p0[0] - nx * 11, p0[1] - ny * 11), (p0[0] + nx * 11, p0[1] + ny * 11)],
                fill=(239, 68, 68, 255),
                width=4
            )
            tdraw.ellipse(
                [p0[0] - 4, p0[1] - 4, p0[0] + 4, p0[1] + 4],
                fill=(16, 185, 129, 255)
            )

    img = Image.alpha_composite(img, track_layer)

    for alias in out_paths:
        out_file = os.path.join(OUTPUT_DIR, f"{alias}.png")
        img.save(out_file, "PNG")


def process_circuit(name, aliases, f1_id, coords_hint, osm_name, force=False):
    existing = [a for a in aliases if os.path.exists(os.path.join(OUTPUT_DIR, f"{a}.png"))]
    if not force and len(existing) == len(aliases):
        print(f"[SKIP] {name} (già presente)")
        return True

    print(f"[SCARICO] {name}...")
    pts = fetch_from_f1_github(f1_id)
    source = "GitHub F1-Circuits"

    if not pts:
        pts = fetch_from_overpass(coords_hint, osm_name)
        source = "OpenStreetMap"

    if pts and len(pts) >= 10:
        render_track_png(pts, aliases)
        print(f"   [OK] {name} ({len(pts)} punti da {source}) -> {', '.join(a + '.png' for a in aliases)}")
        return True
    else:
        print(f"   [FALLITO] Impossibile trovare coordinate GPS per: {name}")
        return False


def main():
    print("=== GENERAZIONE MAPPE CIRCUITI REALI PER EVO RESULTS ===")
    print(f"Cartella destinazione: {OUTPUT_DIR}\n")

    if len(sys.argv) >= 4 and sys.argv[1] == "--aggiungi":
        query_name = sys.argv[2]
        file_alias = sys.argv[3].lower().replace(".png", "").strip()
        aliases = list({file_alias, f"ks_{file_alias}"})
        process_circuit(query_name, aliases, None, None, query_name, force=True)
        return

    force = "--force" in sys.argv
    ok_count = 0
    for name, aliases, f1_id, coords_hint, osm_name in CIRCUITI_KUNOS:
        if process_circuit(name, aliases, f1_id, coords_hint, osm_name, force=force):
            ok_count += 1
        time.sleep(1.2)

    print(f"\nCompletato! {ok_count}/{len(CIRCUITI_KUNOS)} circuiti generati con successo.")


if __name__ == "__main__":
    main()

const nationCache = new Map();

async function lookupDriverNation(name) {
  if (!name) return '';
  const cleanKey = name.trim().toLowerCase();
  if (nationCache.has(cleanKey)) {
    return nationCache.get(cleanKey);
  }
  try {
    const res = await db.execute({
      sql: `SELECT nation FROM laps
            WHERE LOWER(TRIM(driver_name)) = ? AND nation IS NOT NULL AND TRIM(nation) != ''
            ORDER BY id DESC LIMIT 1`,
      args: [cleanKey]
    });
    const nat = res.rows.length > 0 ? String(res.rows[0].nation || '') : '';
    if (nat) nationCache.set(cleanKey, nat);
    return nat;
  } catch (e) {
    return '';
  }
}

function parseDateFromFilename(filename, fallbackDate) {
  const m = filename.match(/(\d{4})(\d{2})(\d{2})_(\d{2})(\d{2})(\d{2})/);
  if (m) {
    return `${m[1]}-${m[2]}-${m[3]} ${m[4]}:${m[5]}:${m[6]}`;
  }
  return fallbackDate;
}

function formatCarName(rawCar) {
  if (!rawCar) return 'GT3';
  return rawCar
    .replace(/^ks_/, '')
    .replace(/_/g, ' ')
    .replace(/\b\w/g, l => l.toUpperCase());
}

const processedHashes = new Set();

async function processResultFile(filePath, serverFolder) {
  const filename = path.basename(filePath);
  const fileHash = `${serverFolder}_${filename}`;

  if (processedHashes.has(fileHash)) return;

  try {
    const checkRes = await db.execute({
      sql: 'SELECT id FROM sessions WHERE file_hash = ?',
      args: [fileHash]
    });
    if (checkRes.rows.length > 0) {
      processedHashes.add(fileHash);
      return;
    }

    const raw = fs.readFileSync(filePath, 'utf-8');
    const data = JSON.parse(raw);

    if (!data.laps || data.laps.length === 0) {
      processedHashes.add(fileHash);
      return;
    }

    const stats = fs.statSync(filePath);
    const playedAt = parseDateFromFilename(filename, stats.mtime.toISOString().replace('T', ' ').substring(0, 19));

    const driversMap = new Map();
    (data.drivers || []).forEach(d => {
      const key = `${d.guid.a}_${d.guid.b}`;
      const fullName = `${d.first_name || ''} ${d.last_name || ''}`.trim() || d.nickname || 'Sconosciuto';
      driversMap.set(key, {
        guid: key,
        name: fullName,
        nickname: d.nickname || '',
        nation: d.nation || ''
      });
      if (d.nation) {
        nationCache.set(fullName.toLowerCase(), d.nation);
      }
    });

    const carsMap = new Map();
    (data.cars || []).forEach(c => {
      const key = `${c.car_id.a}_${c.car_id.b}`;
      carsMap.set(key, {
        model: c.model_displayname || 'Auto Sconosciuta',
        number: c.race_number || 0
      });
    });

    const insertSessRes = await db.execute({
      sql: `INSERT INTO sessions (file_hash, filename, server_name, session_name, session_type, track_name, track_layout, played_at, driver_count)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)`,
      args: [
        fileHash,
        filename,
        data.server_name || serverFolder,
        data.session_name || '',
        data.session_type || 'Practice',
        data.track_name || 'Sconosciuta',
        data.track_layout_name || '',
        playedAt,
        (data.drivers || []).length
      ]
    });

    const sessionId = Number(insertSessRes.lastInsertRowid);

    const driverBestLap = new Map();
    const driverLapCount = new Map();
    const driverCar = new Map();
    const batchStmts = [];

    for (const lap of data.laps) {
      const dKey = `${lap.driver_key.a}_${lap.driver_key.b}`;
      const cKey = `${lap.car_key.a}_${lap.car_key.b}`;
      const driver = driversMap.get(dKey) || { guid: dKey, name: 'Pilota', nation: '' };
      const car = carsMap.get(cKey) || { model: 'GT3', number: 0 };

      const isValid = lap.flags === 2 ? 1 : 0;
      const splits = Array.isArray(lap.split) ? lap.split : [];

      driverCar.set(dKey, car);
      driverLapCount.set(dKey, (driverLapCount.get(dKey) || 0) + 1);

      if (isValid && lap.time > 0) {
        const prevBest = driverBestLap.get(dKey);
        if (!prevBest || lap.time < prevBest) {
          driverBestLap.set(dKey, lap.time);
        }
      }

      batchStmts.push({
        sql: `INSERT INTO laps (session_id, driver_guid, driver_name, nation, car_model, track_name, track_layout, lap_time_ms, sector_1_ms, sector_2_ms, sector_3_ms, is_valid, recorded_at)
              VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
        args: [
          sessionId,
          dKey,
          driver.name,
          driver.nation,
          car.model,
          data.track_name || 'Sconosciuta',
          data.track_layout_name || '',
          lap.time,
          splits[0] || 0,
          splits[1] || 0,
          splits[2] || 0,
          isValid,
          playedAt
        ]
      });
    }

    const standings = data.driver_standings || [];
    const times = data.time_standings || [];

    standings.forEach((st, idx) => {
      const dKey = `${st.a}_${st.b}`;
      const driver = driversMap.get(dKey);
      if (!driver) return;
      const car = driverCar.get(dKey) || { model: '-', number: 0 };
      const lapsDone = driverLapCount.get(dKey) || 0;
      const bestLap = driverBestLap.get(dKey) || 0;

      if (lapsDone > 0) {
        batchStmts.push({
          sql: `INSERT INTO session_results (session_id, position, driver_guid, driver_name, nickname, nation, car_model, race_number, total_time_ms, best_lap_ms, laps_completed)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
          args: [
            sessionId,
            idx + 1,
            dKey,
            driver.name,
            driver.nickname,
            driver.nation,
            car.model,
            car.number,
            times[idx] || 0,
            bestLap,
            lapsDone
          ]
        });
      }
    });

    const CHUNK_SIZE = 100;
    for (let i = 0; i < batchStmts.length; i += CHUNK_SIZE) {
      await db.batch(batchStmts.slice(i, i + CHUNK_SIZE), 'write');
    }

    processedHashes.add(fileHash);
    console.log(`[TURSO IMPORTATO] ${serverFolder} -> ${filename} (${data.track_name} - ${data.session_type})`);
  } catch (err) {
    console.error(`[ERRORE] Lettura/Invio ${filename}:`, err.message);
  }
}

async function parseLiveLog(logPath, isRaceWeekend) {
  let currentPlayers = 0;
  let sessionPhase = isRaceWeekend ? 'QUALIFICHE' : 'PRATICA';
  const activeDrivers = new Map();

  try {
    const stat = fs.statSync(logPath);
    const readSize = Math.min(stat.size, 500000);
    const fd = fs.openSync(logPath, 'r');
    const buffer = Buffer.alloc(readSize);
    fs.readSync(fd, buffer, 0, readSize, stat.size - readSize);
    fs.closeSync(fd);

    const content = buffer.toString('utf-8');
    const lines = content.split(/\r?\n/);

    for (const line of lines) {
      if (line.includes('TimeAttackRemote Qualifying created')) {
        sessionPhase = isRaceWeekend ? 'QUALIFICHE' : 'PRATICA';
      } else if (line.includes('InstantRaceRemote Race created')) {
        sessionPhase = 'GARA';
      }

      const updMatch = line.match(/Server updated:\s*(\d+)\s*players/);
      if (updMatch) {
        currentPlayers = parseInt(updMatch[1], 10);
      }

      const natMatch = line.match(/(\d{17})[^A-Za-z0-9]+(?:nation|country)[=:\s]+"?([A-Za-z]{2,3})"?/i);
      if (natMatch) {
        const [, steamId, nat] = natMatch;
        const existing = activeDrivers.get(steamId) || { steamId, number: '-', name: 'Pilota', car: 'In pista' };
        existing.nation = nat.toUpperCase();
        activeDrivers.set(steamId, existing);
      }

      const carCreateMatch = line.match(/Car \[([^\]]+)\] #(\d+) for driver (.+?) \[(\d+)\]/);
      if (carCreateMatch) {
        const [, carId, number, rawName, steamId] = carCreateMatch;
        const cleanName = rawName.trim();
        const existing = activeDrivers.get(steamId) || {};
        activeDrivers.set(steamId, {
          steamId,
          carId,
          number,
          name: cleanName,
          car: existing.car || 'In pista',
          nation: existing.nation || ''
        });
      }

      const connCarMatch = line.match(/(\d{17}) connected \(true\) on car ([^,]+), with new carId ([^\s]+)/);
      if (connCarMatch) {
        const [, steamId, rawCar, carId] = connCarMatch;
        const existing = activeDrivers.get(steamId) || { steamId, number: '-', name: 'Pilota' };
        existing.car = formatCarName(rawCar);
        existing.carId = carId;
        activeDrivers.set(steamId, existing);
      }

      const destroyMatch = line.match(/Destroying Car\(\d+\) By ConnectionID \(.+?\s+(\d{17})\)/);
      if (destroyMatch) {
        activeDrivers.delete(destroyMatch[1]);
      }

      const discMatch = line.match(/(\d{17})\s+(?:disconnected|timed out)/i);
      if (discMatch) {
        activeDrivers.delete(discMatch[1]);
      }
    }

    if (currentPlayers === 0) {
      activeDrivers.clear();
    } else if (activeDrivers.size > currentPlayers) {
      const entries = Array.from(activeDrivers.entries());
      const trimmed = entries.slice(entries.length - currentPlayers);
      activeDrivers.clear();
      for (const [k, v] of trimmed) {
        activeDrivers.set(k, v);
      }
    }
  } catch (e) {}

  const playersList = [];
  for (const p of activeDrivers.values()) {
    const nat = p.nation || (await lookupDriverNation(p.name));
    playersList.push({ ...p, nation: nat });
  }

  return {
    currentPlayers: playersList.length,
    sessionPhase,
    playersList
  };
}

let isScanning = false;
let lastLiveJson = '';

async function scanServers() {
  if (isScanning) return;
  isScanning = true;

  try {
    if (!fs.existsSync(SERVERS_ROOT)) {
      console.log('Cartella server non trovata:', SERVERS_ROOT);
      return;
    }

    const folders = fs.readdirSync(SERVERS_ROOT).filter(f =>
      fs.statSync(path.join(SERVERS_ROOT, f)).isDirectory()
    );

    const liveStatus = [];

    for (const folder of folders) {
      const serverDir = path.join(SERVERS_ROOT, folder);

      for (const resName of ['Results', 'result']) {
        const resDir = path.join(serverDir, resName);
        if (fs.existsSync(resDir)) {
          const files = fs.readdirSync(resDir).filter(f => f.endsWith('.json'));
          for (const file of files) {
            await processResultFile(path.join(resDir, file), folder);
          }
        }
      }

      let serverName = folder;
      let maxPlayers = 20;
      let track = '';
      let trackId = '';
      let trackLayout = '';
      let isRaceWeekend = false;

      const serverJsonPath = path.join(serverDir, 'server.json');
      if (fs.existsSync(serverJsonPath)) {
        try {
          const sConf = JSON.parse(fs.readFileSync(serverJsonPath, 'utf-8'));
          if (sConf.name) serverName = sConf.name;
          else if (sConf.server_name) serverName = sConf.server_name;
          if (sConf.max_players) maxPlayers = sConf.max_players;
          if (sConf.track) {
            trackId = sConf.track;
            trackLayout = sConf.layout || '';
            track = sConf.layout ? `${sConf.track} (${sConf.layout})` : sConf.track;
          }
          if (sConf.game_type && sConf.game_type.includes('RACE')) {
            isRaceWeekend = true;
          }
        } catch (e) {}
      }

      let isOnline = false;
      let logData = { currentPlayers: 0, sessionPhase: isRaceWeekend ? 'QUALIFICHE' : 'PRATICA', playersList: [] };
      const logsDir = path.join(serverDir, 'Logs');

      if (fs.existsSync(logsDir)) {
        const logFiles = fs.readdirSync(logsDir)
          .filter(f => f.endsWith('.txt') || f.endsWith('.log'))
          .map(f => ({ name: f, time: fs.statSync(path.join(logsDir, f)).mtime.getTime() }))
          .sort((a, b) => b.time - a.time);

        if (logFiles.length > 0) {
          const latestLog = path.join(logsDir, logFiles[0].name);
          const stat = fs.statSync(latestLog);
          isOnline = (Date.now() - stat.mtime.getTime()) < 180000;
          if (isOnline) {
            logData = await parseLiveLog(latestLog, isRaceWeekend);
          }
        }
      }

      const finalCount = isOnline ? logData.playersList.length : 0;

      liveStatus.push({
        id: folder,
        name: serverName,
        online: isOnline,
        players: finalCount,
        maxPlayers,
        track,
        trackId,
        trackLayout,
        sessionPhase: isOnline ? logData.sessionPhase : 'OFFLINE',
        playersList: isOnline ? logData.playersList : []
      });
    }

    const serialized = JSON.stringify(liveStatus, null, 2);
    fs.writeFileSync(path.join(__dirname, 'live_status.json'), serialized);

    // Aggiorna la tabella live_status su Turso (al cambio stato o almeno ogni 45s per mantenere vivo il timestamp)
    const nowIso = new Date().toISOString();
    const shouldPush = serialized !== lastLiveJson || (Date.now() - (scanServers.lastHeartbeat || 0)) > 45000;
    if (shouldPush) {
      await db.execute({
        sql: `INSERT INTO live_status (id, payload, updated_at)
              VALUES (1, ?, ?)
              ON CONFLICT(id) DO UPDATE SET payload = excluded.payload, updated_at = excluded.updated_at`,
        args: [JSON.stringify(liveStatus), nowIso]
      });
      lastLiveJson = serialized;
      scanServers.lastHeartbeat = Date.now();
    }
  } catch (err) {
    console.error('[ERRORE SCAN]', err.message);
  } finally {
    isScanning = false;
  }
}

function setCookie(name: string, value: string, days = 30) {
  if (typeof document === 'undefined') return;
  document.cookie = `${name}=${encodeURIComponent(value)}; expires=${new Date(Date.now() + days * 864e5).toUTCString()}; path=/; SameSite=Lax`;
}

function getCookie(name: string): string | null {
  if (typeof document === 'undefined') return null;
  const m = document.cookie.match(new RegExp('(?:^|; )' + name.replace(/([.$?*|{}()\[\]\\/+^])/g, '\\$1') + '=([^;]*)'));
  return m ? decodeURIComponent(m[1]) : null;
}

function formatMs(ms: number): string {
  if (!ms || ms <= 0) return '--:--.---';
  const m = Math.floor(ms / 60000), s = Math.floor((ms % 60000) / 1000), ml = ms % 1000;
  return `${m}:${s.toString().padStart(2, '0')}.${ml.toString().padStart(3, '0')}`;
}

function formatGap(ms: number, bestMs: number): string {
  if (!ms || !bestMs || ms <= bestMs) return '-';
  return `+${((ms - bestMs) / 1000).toFixed(3)}s`;
}

function formatMs(ms: number): string {
  if (!ms || ms <= 0) return '--:--.---';
  const minutes = Math.floor(ms / 60000);
  const seconds = Math.floor((ms % 60000) / 1000);
  const millis = ms % 1000;
  return `${minutes}:${seconds.toString().padStart(2, '0')}.${millis.toString().padStart(3, '0')}`;
}

function formatGap(ms: number, bestMs: number): string {
  if (!ms || !bestMs || ms <= bestMs) return 'Leader';
  const diff = (ms - bestMs) / 1000;
  return `+${diff.toFixed(3)}s`;
}

export async function POST(request: Request) {
  try {
    await ensureSchema();
    const body = await request.json();
    const message = (body.message || '').trim();

    if (!message) {

        reply: 'Scrivi il tuo nome o cognome (es. "Mario Rossi" oppure "Posizione Rossi") per conoscere la tua posizione e i tuoi tempi migliori.'

    const tracksRes = await db.execute('SELECT DISTINCT track_name FROM laps WHERE is_valid = 1');

    for (const t of tracksRows) {
      const tLow = String(t.track_name || '').toLowerCase();
      const shortName = tLow.replace(/^ks_/, '').replace(/_/g, ' ');
      if (lowerMsg.includes(tLow) || (shortName.length > 3 && lowerMsg.includes(shortName))) {
        matchedTrack = String(t.track_name);

      'record', 'su', 'a', 'in', 'di', 'del', 'della', 'per', 'che', 'ha', 'fatto', 'sono',
      'quanto', 'sto', 'pista', 'circuito', 'giri', 'veloci', 'nel', 'sito', 'stats', 'statistiche'
    ];

      .replace(/[?!.,;:"'()]/g, ' ')

        reply: 'Non ho capito il nome del pilota. Scrivi direttamente il nome o il cognome (es. "Rossi")!'

      ? matchedDrivers
      : allDrivers.filter(d => {
          const dLow = String(d.driver_name || '').toLowerCase();
          return tokens.some((tok: string) => dLow.includes(tok));
        });

    if (finalDrivers.length === 0) {
      return NextResponse.json({
        reply: `Nessun pilota trovato con "${tokens.join(' ')}". Verifica di aver registrato almeno un giro valido sui server.`
      });
    }

    if (finalDrivers.length > 3) {
      const namesList = finalDrivers.slice(0, 6).map(d => d.driver_name).join(', ');
      return NextResponse.json({
        reply: `Ho trovato diversi piloti (${namesList}...). Specifica meglio nome e cognome!`
      });
    }

    const responses: string[] = [];

    for (const driver of finalDrivers) {
      let trackList: { track_name: string }[] = [];
      if (matchedTrack) {
        trackList = [{ track_name: matchedTrack }];
      } else {
        const driverTracksRes = await db.execute({
          sql: `SELECT DISTINCT track_name
                FROM laps
                WHERE driver_guid = ? AND is_valid = 1 AND lap_time_ms > 0
                ORDER BY track_name ASC`,
          args: [String(driver.driver_guid)]
        });
        trackList = driverTracksRes.rows as unknown as { track_name: string }[];
      }

      if (trackList.length === 0) {
        responses.push(`Nessun giro valido trovato per **${driver.driver_name}** su questa pista.`);
        continue;
      }

      const lines: string[] = [`🏁 **${driver.driver_name}** — Riepilogo Giri Veloci:`];

      for (const tr of trackList) {
        const standingsRes = await db.execute({
          sql: `WITH RankedDriverLaps AS (
                  SELECT
                    driver_guid,
                    driver_name,
                    car_model,
                    lap_time_ms,
                    sector_1_ms,
                    sector_2_ms,
                    sector_3_ms,
                    ROW_NUMBER() OVER (PARTITION BY driver_guid ORDER BY lap_time_ms ASC) as rn
                  FROM laps
                  WHERE track_name = ? AND is_valid = 1 AND lap_time_ms > 0
                )
                SELECT driver_guid, driver_name, car_model, lap_time_ms, sector_1_ms, sector_2_ms, sector_3_ms
                FROM RankedDriverLaps
                WHERE rn = 1
                ORDER BY lap_time_ms ASC`,
          args: [String(tr.track_name)]
        });
        const standings = standingsRes.rows as any[];

        const posIndex = standings.findIndex(s => String(s.driver_guid) === String(driver.driver_guid));
        if (posIndex !== -1) {
          const entry = standings[posIndex];
          const leader = standings[0];
          const pos = posIndex + 1;
          const totalDrivers = standings.length;
          const entryMs = Number(entry.lap_time_ms);
          const leaderMs = Number(leader.lap_time_ms);

          const gapText = pos === 1
            ? (standings.length > 1 ? `1° Assoluto (-${((Number(standings[1].lap_time_ms) - entryMs) / 1000).toFixed(3)}s sul 2°)` : '1° Assoluto')
            : `${formatGap(entryMs, leaderMs)} dal 1°`;

          lines.push(
            `• **${tr.track_name}**: **#${pos}** su ${totalDrivers} piloti — Tempo: **${formatMs(entryMs)}** (${gapText}) | Auto: ${entry.car_model}`
          );
        }
      }

      responses.push(lines.join('\n'));
    }

    return NextResponse.json({
      reply: responses.join('\n\n')
    });
  } catch (err) {
    console.error('Errore API Chat:', err);
    return NextResponse.json({
      reply: 'Si è verificato un errore durante la ricerca dei tempi nel database.'
    }, { status: 500 });
  }
}

import { createClient, Client } from '@libsql/client';

let client: Client | null = null;

export function getDb(): Client {
  if (!client) {
    const url = process.env.TURSO_DATABASE_URL;
    const authToken = process.env.TURSO_AUTH_TOKEN;
    if (!url) {
      throw new Error('TURSO_DATABASE_URL non configurato nelle variabili d ambiente');
    }
    client = createClient({
      url,
      authToken,
    });
  }
  return client;
}

const db = {
  execute: (...args: Parameters<Client['execute']>) => getDb().execute(...args),
  batch: (...args: Parameters<Client['batch']>) => getDb().batch(...args),
};

export default db;

let schemaInitialized = false;

export async function ensureSchema() {
  if (schemaInitialized) return;
  const conn = getDb();

  await conn.batch([
    `CREATE TABLE IF NOT EXISTS sessions (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      file_name TEXT UNIQUE,
      server_name TEXT,
      track_name TEXT,
      session_type TEXT,
      date_time TEXT,
      winner_name TEXT,
      winner_car TEXT,
      total_drivers INTEGER
    )`,
    `CREATE TABLE IF NOT EXISTS session_results (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      session_id INTEGER,
      position INTEGER,
      driver_name TEXT,
      driver_guid TEXT,
      nation TEXT,
      car_model TEXT,
      best_lap_ms INTEGER,
      total_time_ms INTEGER,
      laps_count INTEGER,
      grid_position INTEGER,
      FOREIGN KEY(session_id) REFERENCES sessions(id)
    )`,
    `CREATE TABLE IF NOT EXISTS laps (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      session_id INTEGER,
      driver_name TEXT,
      driver_guid TEXT,
      nation TEXT,
      car_model TEXT,
      track_name TEXT,
      lap_time_ms INTEGER,
      sector1_ms INTEGER,
      sector2_ms INTEGER,
      sector3_ms INTEGER,
      cuts INTEGER,
      timestamp TEXT,
      FOREIGN KEY(session_id) REFERENCES sessions(id)
    )`,
    `CREATE TABLE IF NOT EXISTS live_status (
      id INTEGER PRIMARY KEY CHECK (id = 1),
      payload TEXT,
      updated_at INTEGER
    )`
  ], 'write');

  schemaInitialized = true;
}
